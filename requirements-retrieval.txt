#   pip install -r requirements-retrieval.txt

# ⚠️ 第一行不能省。Windows 上 PyPI 默认给的 torch wheel 是 **CUDA 版**，
#    约 2.4GB，而本项目的向量模型跑在 CPU 上（7500 条 chunk 编码约 1~2 分钟，
#    检索是纯矩阵乘法，用不到 GPU）。走 PyTorch 官方 CPU 源只要约 200MB。
#    先装 CPU 版 torch 之后，下面再写 torch 也没有依赖会把它换回 CUDA 版。
# ============================================================

# --- BM25 稀疏检索 ---
jieba>=0.42.1              # 中文分词（BM25 用）
zhconv>=1.4                # 简繁归一（纯 Python，很小）
#
# zhconv 不能省。10 家里有 5 家是港股年报（中国人保、中国太平、中国再保险、
# 中国财险、众安在线），正文是繁体，占全部 chunk 的 52%。不转换的话，
# 简体提问"总资产"永远匹配不上原文的"總資產"，这几家会像坏掉一样查不到。
# 详见 src/tokenize.py 开头的说明。

# --- 向量检索 ---
# torch 由上面那行单独安装（CPU 版），这里不重复声明
sentence-transformers>=3.0 # 中文 embedding 模型
numpy>=1.26                # 向量存盘 + 余弦相似度

# 注意：**不需要 faiss**。本项目 7500 条 chunk × 512 维 ≈ 15MB，
# 直接 numpy 矩阵点积（一次 7500×512 的乘法）在 CPU 上不到 1ms，
# 比维护一个 faiss 索引更省事，也少一个要编译的依赖。

# 注意：**不需要 rank-bm25**。BM25 是自己实现的（src/bm25.py），原因见该文件
# 开头的注释 —— 需要在打分前按公司做布尔过滤，通用库不提供这个钩子。

# 建议的中文向量模型（首次运行会自动下载）：
#   - BAAI/bge-small-zh-v1.5   （本项目在用，512 维，约 95MB）
#   - BAAI/bge-large-zh-v1.5   （效果更好，约 1.3GB，CPU 编码会慢很多）
#   - moka-ai/m3e-base         （金融语料上表现也不错）
#
